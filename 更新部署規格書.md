# linyize1111.github.io 網站更新規格書 (Deployment Specification)

這份文件記錄了如何將本次為您的個人網站所開發的新功能，**完整且精確地同步到另一部電腦上**。
> ⚠️ **注意**：根據您的要求，本規格書**不會包含**首頁 (`index.html`) 兩張卡片的 CSS 排版修改（保留原主題預設），只專注於資料、內容、社群連結與分頁功能的升級。

## 步驟七：替換 JavaScript 核心排序與分頁邏輯
在 `directory.html` 與 `literature.html` 的最底部，找到原本的 `<!-- 📂 文章排序與篩選邏輯 -->` 的 `<script>` 區塊，將整個區塊**完全替換**為以下具備分頁（一頁四筆）功能的強大新腳本：

```html
<!-- 📂 文章排序、篩選與分頁邏輯 -->
<script>
    document.addEventListener('DOMContentLoaded', function () {
        const filterCategory = document.getElementById('filter-category');
        const sortBy = document.getElementById('sort-by');
        const container = document.getElementById('posts-container');
        const paginationControls = document.getElementById('pagination-controls');
        const pageInfo = document.getElementById('page-info');
        const allItems = Array.from(container.getElementsByClassName('note-item'));
        const initialOrder = allItems.slice();
        const ITEMS_PER_PAGE = 4;
        let currentPage = 1;
        let activeItems = []; // 當前篩選+排序後的所有項目

        function getFilteredSorted() {
            const category = filterCategory.value;
            const sortType = sortBy.value;

            let filtered = allItems.filter(item => {
                if (category === 'all') return true;
                return item.dataset.category === category;
            });

            if (sortType !== 'default') {
                filtered.sort((a, b) => {
                    if (sortType === 'upload-desc') return new Date(b.dataset.upload) - new Date(a.dataset.upload);
                    if (sortType === 'upload-asc')  return new Date(a.dataset.upload) - new Date(b.dataset.upload);
                    if (sortType === 'edit-desc')   return new Date(b.dataset.edit)   - new Date(a.dataset.edit);
                    if (sortType === 'title-asc')   return a.dataset.title.localeCompare(b.dataset.title, 'zh-TW');
                    if (sortType === 'title-desc')  return b.dataset.title.localeCompare(a.dataset.title, 'zh-TW');
                    return 0;
                });
            } else {
                filtered = initialOrder.filter(item => filtered.includes(item));
            }
            return filtered;
        }

        function renderPage() {
            const totalPages = Math.max(1, Math.ceil(activeItems.length / ITEMS_PER_PAGE));
            if (currentPage > totalPages) currentPage = totalPages;

            const start = (currentPage - 1) * ITEMS_PER_PAGE;
            const end   = start + ITEMS_PER_PAGE;

            // 隱藏所有項目，再顯示當頁項目，並保持 DOM 順序
            allItems.forEach(item => item.style.display = 'none');
            activeItems.forEach((item, i) => {
                item.style.display = (i >= start && i < end) ? '' : 'none';
                container.appendChild(item);
            });

            // 更新頁碼資訊
            pageInfo.textContent = activeItems.length > 0
                ? `第 ${currentPage} 頁，共 ${totalPages} 頁（${activeItems.length} 篇文章）`
                : '無符合條件的內容';

            // 建立分頁按鈕
            paginationControls.innerHTML = '';
            if (totalPages <= 1) return;

            function makeBtn(label, page, disabled, isActive) {
                const btn = document.createElement('button');
                btn.textContent = label;
                if (disabled) btn.disabled = true;
                if (isActive) btn.classList.add('active');
                btn.addEventListener('click', function () {
                    currentPage = page;
                    renderPage();
                });
                return btn;
            }

            paginationControls.appendChild(makeBtn('«', 1, currentPage === 1, false));
            paginationControls.appendChild(makeBtn('‹', currentPage - 1, currentPage === 1, false));

            // 顯示頁碼按鈕（最多展現 5 頁）
            let startPage = Math.max(1, currentPage - 2);
            let endPage   = Math.min(totalPages, startPage + 4);
            if (endPage - startPage < 4) startPage = Math.max(1, endPage - 4);

            for (let p = startPage; p <= endPage; p++) {
                paginationControls.appendChild(makeBtn(p, p, false, p === currentPage));
            }

            paginationControls.appendChild(makeBtn('›', currentPage + 1, currentPage === totalPages, false));
            paginationControls.appendChild(makeBtn('»', totalPages, currentPage === totalPages, false));
        }

        function updateView() {
            currentPage = 1;
            activeItems = getFilteredSorted();
            renderPage();
        }

        filterCategory.addEventListener('change', updateView);
        sortBy.addEventListener('change', updateView);

        // 初始化
        updateView();
    });
</script>
```
